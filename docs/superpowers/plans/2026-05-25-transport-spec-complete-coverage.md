# Transport Spec v0.11 — Complete Coverage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extend the `.tis` transport protocol to round-trip every `SpectralDataset` accessor with byte-equivalent content fidelity, across Java/Python/ObjC, plus a conformance test architecture that catches future "silent drop" regressions.

**Architecture:** 12 new packet types (`0x10`–`0x1B`) covering references, image cubes, identifications, quantifications, dataset-level provenance, subject/sample metadata, and the dataset-level encryption algorithm name. Tabular packets carry Apache Arrow IPC streams; image cubes use one packet per pixel; references emit `(header, N×chromosome, end)` triples. Backward-compat via length-prefixed wire frames + skip-unknown reader logic, gated by a `transport_v0_11` StreamHeader feature flag.

**Tech Stack:** Java 22 + Maven, Python 3.12 + pip, Objective-C/GNUstep + libobjc2, Apache Arrow C++ 16.0 (libarrow), Arrow Java 16, pyarrow 16, ARROW_OBJC bindings (we'll vendor a minimal Arrow IPC reader/writer for ObjC since there's no official ObjC binding).

import org.junit.jupiter.api.Test;
import java.io.ByteArrayOutputStream;
import java.nio.ByteBuffer;
import java.nio.ByteOrder;
import static org.junit.jupiter.api.Assertions.*;

class TransportReaderSkipUnknownTest {

    @Test
    void unknown_packet_type_is_skipped_not_thrown() throws Exception {
        // Craft a stream: StreamHeader, an unknown-type packet, EndOfStream.
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        TransportWriter w = new TransportWriter(out);
        w.writeStreamHeader("1.2", "test", "",
            java.util.List.of("transport_v0_11"), 0);

        // Manually write a packet with an unknown wire byte (0x7E).
        byte[] payload = "future-extension-data".getBytes();
        PacketHeader unknown = new PacketHeader(
            (byte) 0x7E,    // unknown type
            (byte) 0,       // no flags
            0,              // dataset_id
            0L,             // au_sequence
            payload.length,
            System.currentTimeMillis() * 1_000_000L);
        out.write(unknown.encode());
        out.write(payload);

        w.writeEndOfStream();

        // Reader must consume the unknown packet's header + payload and
        // continue to EndOfStream without raising.
        TransportReader r = new TransportReader(out.toByteArray());
        // Per existing API, materializeTo writes a .tio; here we just
        // assert no exception is thrown while reading the unknown packet.
        java.util.List<TransportReader.PacketRecord> records = r.recordsForTest();
        assertEquals(3, records.size());
        assertEquals(0x7E, records.get(1).header.packetType & 0xFF);
    }
}

import global.thalion.ttio.Identification;
import global.thalion.ttio.Quantification;

import org.apache.arrow.vector.*;
import org.apache.arrow.vector.complex.MapVector;

import org.apache.arrow.vector.ipc.ArrowStreamWriter;
import org.apache.arrow.vector.types.Types;
import org.apache.arrow.vector.types.pojo.*;

import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.nio.channels.Channels;
import java.util.*;

/** Encodes/decodes tabular dataset content (identifications,
 *  quantifications, subjects, samples) as Apache Arrow IPC streams
 *  for transport-spec v0.11 packets 0x16, 0x17, 0x19, 0x1A. */
public final class ArrowIpcCodec {

    private static final RootAllocator ALLOC = new RootAllocator();

    private ArrowIpcCodec() {}

    // ---------- Identifications ----------

    private static final Schema IDENTIFICATION_SCHEMA = new Schema(List.of(
        new Field("peptide",            FieldType.nullable(new ArrowType.Utf8()),   null),
        new Field("protein_accession",  FieldType.nullable(new ArrowType.Utf8()),   null),
        new Field("spectrum_ref",       FieldType.nullable(new ArrowType.Utf8()),   null),
        new Field("score",              FieldType.nullable(new ArrowType.FloatingPoint(
                                            org.apache.arrow.vector.types.FloatingPointPrecision.DOUBLE)), null),
        new Field("extras_json",        FieldType.nullable(new ArrowType.Utf8()),   null)
    ));

    public static byte[] encodeIdentifications(List<Identification> rows) {
        try (VectorSchemaRoot root = VectorSchemaRoot.create(IDENTIFICATION_SCHEMA, ALLOC);
             ByteArrayOutputStream buf = new ByteArrayOutputStream();
             ArrowStreamWriter writer = new ArrowStreamWriter(root, null, Channels.newChannel(buf))) {

            root.allocateNew();
            VarCharVector peptide = (VarCharVector) root.getVector("peptide");
            VarCharVector protein = (VarCharVector) root.getVector("protein_accession");
            VarCharVector spec    = (VarCharVector) root.getVector("spectrum_ref");
            Float8Vector score    = (Float8Vector) root.getVector("score");
            VarCharVector extras  = (VarCharVector) root.getVector("extras_json");

            for (int i = 0; i < rows.size(); i++) {
                Identification r = rows.get(i);
                peptide.setSafe(i, r.peptide().getBytes());
                protein.setSafe(i, r.proteinAccession().getBytes());
                spec.setSafe(i, r.spectrumRef().getBytes());
                score.setSafe(i, r.score());
                extras.setSafe(i, jsonOf(r.extras()).getBytes());
            }
            root.setRowCount(rows.size());

            writer.start();
            writer.writeBatch();
            writer.end();
            return buf.toByteArray();
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }

    public static List<Identification> decodeIdentifications(byte[] ipc) {
        if (ipc == null || ipc.length == 0) return List.of();
        try (ByteArrayInputStream in = new ByteArrayInputStream(ipc);
             ArrowStreamReader reader = new ArrowStreamReader(in, ALLOC)) {
            List<Identification> out = new ArrayList<>();
            VectorSchemaRoot root = reader.getVectorSchemaRoot();
            while (reader.loadNextBatch()) {
                int n = root.getRowCount();
                VarCharVector peptide = (VarCharVector) root.getVector("peptide");
                VarCharVector protein = (VarCharVector) root.getVector("protein_accession");
                VarCharVector spec    = (VarCharVector) root.getVector("spectrum_ref");
                Float8Vector score    = (Float8Vector) root.getVector("score");
                VarCharVector extras  = (VarCharVector) root.getVector("extras_json");

                        new String(peptide.get(i)),
                        new String(protein.get(i)),
                        new String(spec.get(i)),
                        score.get(i),
                        parseJsonMap(new String(extras.get(i)))));

            return out;
        } catch (IOException e) {
            throw new RuntimeException(e);
        }

        new Field("feature",   FieldType.nullable(new ArrowType.Utf8()),   null),
        new Field("sample",    FieldType.nullable(new ArrowType.Utf8()),   null),
        new Field("value",     FieldType.nullable(new ArrowType.FloatingPoint(

        new Field("extras_json", FieldType.nullable(new ArrowType.Utf8()), null)

    public static byte[] encodeQuantifications(List<Quantification> rows) {

        // See identifications method above for the full template.
        try (VectorSchemaRoot root = VectorSchemaRoot.create(QUANTIFICATION_SCHEMA, ALLOC);

             ArrowStreamWriter writer = new ArrowStreamWriter(root, null, Channels.newChannel(buf))) {
            root.allocateNew();

            VarCharVector sample   = (VarCharVector) root.getVector("sample");
            Float8Vector  value    = (Float8Vector)  root.getVector("value");
            VarCharVector metric   = (VarCharVector) root.getVector("metric");

        if (ipc == null || ipc.length == 0) return List.of();

            List<Quantification> out = new ArrayList<>();

                VarCharVector extras   = (VarCharVector) root.getVector("extras_json");

    // ---------- JSON helpers (extras Map<String, Object> via JSON) ----------

    private static String jsonOf(Map<String, Object> extras) {
        if (extras == null || extras.isEmpty()) return "{}";
        StringBuilder sb = new StringBuilder("{");
        boolean first = true;
        for (Map.Entry<String, Object> e : extras.entrySet()) {
            if (!first) sb.append(',');
            first = false;
            sb.append('"').append(e.getKey()).append("\":");
            Object v = e.getValue();
            if (v instanceof Number || v instanceof Boolean) sb.append(v);
            else if (v == null) sb.append("null");
            else sb.append('"').append(v.toString().replace("\"", "\\\"")).append('"');
        }
        sb.append('}');
        return sb.toString();
    }

    @SuppressWarnings("unchecked")
    private static Map<String, Object> parseJsonMap(String s) {
        if (s == null || s.isBlank() || s.equals("{}")) return Map.of();
        // Reuse the existing JSON parser already used by the codebase
        // (e.g. global.thalion.ttio.workbench.transport.WorkbenchHttp uses
        // org.json). Adapt to the actual parser available.
        return (Map<String, Object>) new org.json.JSONObject(s).toMap();
    }

```java
/** Emit a v0.11 reference group: REFERENCE_GROUP_HEADER + N×REFERENCE_CHROMOSOME
 *  + END_OF_REFERENCE_GROUP. */
public void writeReferenceGroup(ReferenceImport ref) throws IOException {
    // -- header --
    ByteArrayOutputStream headerPayload = new ByteArrayOutputStream();
    byte[] uri = ref.uri().getBytes(StandardCharsets.UTF_8);
    writeUint16(headerPayload, uri.length);
    headerPayload.write(uri);
    writeUint32(headerPayload, ref.chromosomes().size());
    writeUint64(headerPayload, totalBasesOf(ref));
    headerPayload.write(ref.md5Hex().getBytes(StandardCharsets.US_ASCII));  // 32 bytes
    writePacket(PacketType.REFERENCE_GROUP_HEADER, 0, 0L,
        headerPayload.toByteArray());

    // -- chromosomes --
    for (int i = 0; i < ref.chromosomes().size(); i++) {
        String name = ref.chromosomes().get(i);
        byte[] seq  = ref.sequences().get(i);
        ByteArrayOutputStream chromPayload = new ByteArrayOutputStream();
        byte[] nameUtf8 = name.getBytes(StandardCharsets.UTF_8);
        writeUint16(chromPayload, nameUtf8.length);
        chromPayload.write(nameUtf8);
        writeUint64(chromPayload, seq.length);
        // Encoding choice: skip compression for small sequences (mirrors
        // ReferenceImport.writeToDataset Perf A).
        if (seq.length < 4096) {
            chromPayload.write(0);  // encoding = 0 = uncompressed UINT8
            writeUint32(chromPayload, seq.length);
            chromPayload.write(seq);
        } else {
            chromPayload.write(1);  // encoding = 1 = zlib
            byte[] compressed = zlibCompress(seq);
            writeUint32(chromPayload, compressed.length);
            chromPayload.write(compressed);
        }
        writePacket(PacketType.REFERENCE_CHROMOSOME, 0, i,
            chromPayload.toByteArray());
    }

    // -- end-of-group --
    ByteArrayOutputStream tailPayload = new ByteArrayOutputStream();
    writeUint32(tailPayload, ref.chromosomes().size());
    writePacket(PacketType.END_OF_REFERENCE_GROUP, 0, 0L,
        tailPayload.toByteArray());
}

private static long totalBasesOf(ReferenceImport ref) {
    long total = 0;
    for (byte[] seq : ref.sequences()) total += seq.length;
    return total;
}

private static byte[] zlibCompress(byte[] in) throws IOException {
    ByteArrayOutputStream baos = new ByteArrayOutputStream();
    try (java.util.zip.DeflaterOutputStream def = new java.util.zip.DeflaterOutputStream(baos)) {
        def.write(in);
    }
    return baos.toByteArray();
}

private static void writeUint16(ByteArrayOutputStream out, int v) {
    out.write((v >>> 8) & 0xFF);
    out.write(v & 0xFF);
}
private static void writeUint32(ByteArrayOutputStream out, long v) {
    out.write((int)((v >>> 24) & 0xFF));
    out.write((int)((v >>> 16) & 0xFF));
    out.write((int)((v >>> 8)  & 0xFF));
    out.write((int)( v         & 0xFF));
}
private static void writeUint64(ByteArrayOutputStream out, long v) {
    for (int i = 7; i >= 0; i--) out.write((int)((v >>> (i*8)) & 0xFF));
}

    ByteBuffer bb = ByteBuffer.wrap(payload).order(ByteOrder.BIG_ENDIAN);
    int nameLen = bb.getShort() & 0xFFFF;

"""Apache Arrow IPC encode/decode for transport v0.11 tabular packets.

Mirrors Java's ArrowIpcCodec for cross-language byte-equality of the
IPC payload (Arrow IPC is endian-independent and the schema is
declarative, so Java + Python emit identical bytes for the same rows).
"""
from __future__ import annotations
import io
import json
from typing import List

import pyarrow as pa
import pyarrow.ipc

from ttio.identifications import Identification
from ttio.quantifications import Quantification


_ID_SCHEMA = pa.schema([
    pa.field("peptide",           pa.string()),
    pa.field("protein_accession", pa.string()),
    pa.field("spectrum_ref",      pa.string()),
    pa.field("score",             pa.float64()),
    pa.field("extras_json",       pa.string()),
])

_Q_SCHEMA = pa.schema([
    pa.field("feature",     pa.string()),
    pa.field("sample",      pa.string()),
    pa.field("value",       pa.float64()),
    pa.field("metric",      pa.string()),
    pa.field("extras_json", pa.string()),
])


def encode_identifications(rows: List[Identification]) -> bytes:
    arr = {
        "peptide":           [r.peptide for r in rows],
        "protein_accession": [r.protein_accession for r in rows],
        "spectrum_ref":      [r.spectrum_ref for r in rows],
        "score":             [r.score for r in rows],
        "extras_json":       [_json(r.extras) for r in rows],
    }
    batch = pa.RecordBatch.from_pydict(arr, schema=_ID_SCHEMA)
    sink = io.BytesIO()
    with pa.ipc.new_stream(sink, _ID_SCHEMA) as writer:
        writer.write_batch(batch)
    return sink.getvalue()


def decode_identifications(ipc: bytes) -> List[Identification]:
    if not ipc:
        return []
    reader = pa.ipc.open_stream(io.BytesIO(ipc))
    out: List[Identification] = []
    for batch in reader:
        cols = batch.to_pydict()
        for i in range(batch.num_rows):
            out.append(Identification(
                peptide=cols["peptide"][i],
                protein_accession=cols["protein_accession"][i],
                spectrum_ref=cols["spectrum_ref"][i],
                score=cols["score"][i],
                extras=json.loads(cols["extras_json"][i] or "{}"),
            ))
    return out


def encode_quantifications(rows: List[Quantification]) -> bytes:
    arr = {
        "feature":     [r.feature for r in rows],
        "sample":      [r.sample for r in rows],
        "value":       [r.value for r in rows],
        "metric":      [r.metric for r in rows],
        "extras_json": [_json(r.extras) for r in rows],
    }
    batch = pa.RecordBatch.from_pydict(arr, schema=_Q_SCHEMA)
    sink = io.BytesIO()
    with pa.ipc.new_stream(sink, _Q_SCHEMA) as writer:
        writer.write_batch(batch)
    return sink.getvalue()


def decode_quantifications(ipc: bytes) -> List[Quantification]:
    if not ipc:
        return []
    reader = pa.ipc.open_stream(io.BytesIO(ipc))
    out: List[Quantification] = []
    for batch in reader:
        cols = batch.to_pydict()
        for i in range(batch.num_rows):
            out.append(Quantification(
                feature=cols["feature"][i],
                sample=cols["sample"][i],
                value=cols["value"][i],
                metric=cols["metric"][i],
                extras=json.loads(cols["extras_json"][i] or "{}"),
            ))
    return out


def _json(extras) -> str:
    if not extras:
        return "{}"
    return json.dumps(extras, sort_keys=True, separators=(",", ":"))
```

#import "TTIOArrowIpcCodec.h"
#import "TTIOIdentification.h"
#import "TTIOQuantification.h"

extern NSData *TTIOArrowIpcEncode(NSString *, NSString *);
extern NSString *TTIOArrowIpcDecode(NSString *, NSData *);

@implementation TTIOArrowIpcCodec

+ (NSData *)encodeIdentifications:(NSArray<TTIOIdentification *> *)rows {
    NSMutableArray *json = [NSMutableArray new];
    for (TTIOIdentification *r in rows) {
        [json addObject:@{
            @"peptide": r.peptide ?: @"",
            @"protein_accession": r.proteinAccession ?: @"",
            @"spectrum_ref": r.spectrumRef ?: @"",
            @"score": @(r.score),
            @"extras_json": r.extras
                ? [NSString stringWithUTF8String:
                    [[NSJSONSerialization dataWithJSONObject:r.extras
                        options:NSJSONWritingSortedKeys error:nil] bytes]]
                : @"{}",
        }];
    }
    NSData *jsonBytes = [NSJSONSerialization dataWithJSONObject:json
        options:NSJSONWritingSortedKeys error:nil];
    return TTIOArrowIpcEncode(@"identifications",
        [[NSString alloc] initWithData:jsonBytes encoding:NSUTF8StringEncoding]);
}

+ (NSArray<TTIOIdentification *> *)decodeIdentifications:(NSData *)ipc {
    if (!ipc || ipc.length == 0) return @[];
    NSString *rowsJson = TTIOArrowIpcDecode(@"identifications", ipc);
    NSArray *raw = [NSJSONSerialization JSONObjectWithData:
        [rowsJson dataUsingEncoding:NSUTF8StringEncoding]
        options:0 error:nil];
    NSMutableArray *out = [NSMutableArray new];
    for (NSDictionary *row in raw) {
        TTIOIdentification *id = [TTIOIdentification new];
        id.peptide = row[@"peptide"];
        id.proteinAccession = row[@"protein_accession"];
        id.spectrumRef = row[@"spectrum_ref"];
        id.score = [row[@"score"] doubleValue];
        NSString *extras = row[@"extras_json"] ?: @"{}";
        id.extras = [NSJSONSerialization JSONObjectWithData:
            [extras dataUsingEncoding:NSUTF8StringEncoding] options:0 error:nil];
        [out addObject:id];
    }
    return out;
}

// + encodeQuantifications: / + decodeQuantifications: follow the same pattern.

@end
